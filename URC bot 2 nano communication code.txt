// =====================================================
// NANO 1 - MASTER
//
// MODE 1 = LINE FOLLOWER
// MODE 2 = MAZE
// MODE 3 = RC MANUAL
//
// BUTTON 1 -> SELECT MODE
// BUTTON 2 -> START
//
// LED1 ON + LED2 OFF = LINE FOLLOWER
// LED1 OFF + LED2 ON = MAZE
// LED1 ON + LED2 ON  = RC MANUAL
//
// RX D0 <- NANO 2 TX D1
// =====================================================


// =====================================================
// TB6612FNG
// =====================================================

#define PWMB 9
#define PWMA 3

#define BIN1 7
#define BIN2 8

#define AIN1 5
#define AIN2 4

#define STDBY 6


// =====================================================
// BUTTONS / LEDs
// =====================================================

#define led1 11
#define led2 12

#define Button1 2
#define Button2 10


// =====================================================
// MODES
// =====================================================

#define LINE_MODE 0
#define MAZE_MODE 1
#define RC_MODE   2

int mode = LINE_MODE;


// =====================================================
// SPEED
// +20 FROM PREVIOUS CODE
// =====================================================

int robotSpeed = 120;

int forwardSpeed = 120;
int backwardSpeed = 120;

int calibrationSpeed = 120;
int uturnSpeed = 100;


// =====================================================
// PID
// =====================================================

int kp = 30;
int kd = 120;
int ki = 0;


// =====================================================
// IR
// =====================================================

int P, D, I;
int lastError = 0;
int PIDoutput;

int error;
int speedA;
int speedB;
int sensorCount;

bool lineFound = true;

int weights[8] =
{
  -8, -4, -2, -1,
   1,  2,  4,  8
};

int minimum[8] =
{
  1023,1023,1023,1023,
  1023,1023,1023,1023
};

int maximum[8] =
{
  0,0,0,0,
  0,0,0,0
};

int analog[8];
int binary[8];


// =====================================================
// NANO 2 DATA
// =====================================================

int leftDistance = 400;
int centerDistance = 400;
int rightDistance = 400;

int rcSteering = 1500;
int rcThrottle = 1500;

int obstacleDistance = 15;


// =====================================================
// SETUP
// =====================================================

void setup()
{
  pinMode(PWMA, OUTPUT);
  pinMode(PWMB, OUTPUT);

  pinMode(AIN1, OUTPUT);
  pinMode(AIN2, OUTPUT);

  pinMode(BIN1, OUTPUT);
  pinMode(BIN2, OUTPUT);

  pinMode(STDBY, OUTPUT);

  pinMode(led1, OUTPUT);
  pinMode(led2, OUTPUT);

  pinMode(Button1, INPUT_PULLUP);
  pinMode(Button2, INPUT_PULLUP);

  digitalWrite(STDBY, HIGH);

  motorControl(0, 0);

  Serial.begin(9600);

  showMode();
}


// =====================================================
// MAIN LOOP
// =====================================================

void loop()
{
  receiveData();


  // ===================================================
  // BUTTON 1 = CHANGE MODE
  // ===================================================

  if (digitalRead(Button1) == LOW)
  {
    delay(50);

    if (digitalRead(Button1) == LOW)
    {
      mode++;

      if (mode > RC_MODE)
        mode = LINE_MODE;

      showMode();

      while (digitalRead(Button1) == LOW)
      {
        receiveData();
      }

      delay(150);
    }
  }


  // ===================================================
  // BUTTON 2 = START
  // ===================================================

  if (digitalRead(Button2) == LOW)
  {
    delay(50);

    if (digitalRead(Button2) == LOW)
    {
      while (digitalRead(Button2) == LOW)
      {
        receiveData();
      }

      delay(300);

      if (mode == LINE_MODE)
        lineFollower();

      else if (mode == MAZE_MODE)
        maze();

      else if (mode == RC_MODE)
        rcManual();
    }
  }
}


// =====================================================
// SHOW CURRENT MODE
// =====================================================

void showMode()
{
  if (mode == LINE_MODE)
  {
    digitalWrite(led1, HIGH);
    digitalWrite(led2, LOW);
  }

  else if (mode == MAZE_MODE)
  {
    digitalWrite(led1, LOW);
    digitalWrite(led2, HIGH);
  }

  else
  {
    digitalWrite(led1, HIGH);
    digitalWrite(led2, HIGH);
  }
}


// =====================================================
// RECEIVE DATA FROM NANO 2
//
// FORMAT:
//
// L,C,R,S,T
//
// L = left ultrasonic
// C = center ultrasonic
// R = right ultrasonic
// S = steering PWM
// T = throttle PWM
// =====================================================

void receiveData()
{
  if (Serial.available())
  {
    String data = Serial.readStringUntil('\n');

    int c1 = data.indexOf(',');
    int c2 = data.indexOf(',', c1 + 1);
    int c3 = data.indexOf(',', c2 + 1);
    int c4 = data.indexOf(',', c3 + 1);

    if (
      c1 > 0 &&
      c2 > c1 &&
      c3 > c2 &&
      c4 > c3
    )
    {
      leftDistance =
        data.substring(0, c1).toInt();

      centerDistance =
        data.substring(c1 + 1, c2).toInt();

      rightDistance =
        data.substring(c2 + 1, c3).toInt();

      rcSteering =
        data.substring(c3 + 1, c4).toInt();

      rcThrottle =
        data.substring(c4 + 1).toInt();
    }
  }
}


// =====================================================
// MODE 1
// LINE FOLLOWER
// =====================================================

void lineFollower()
{
  showMode();

  calibrate();

  delay(300);

  while (true)
  {
    readSensors();

    if (lineFound)
    {
      movePID();
    }
    else
    {
      if (error > 0)
        turnLeft();

      else if (error < 0)
        turnRight();

      else
        uTurn();
    }
  }
}


// =====================================================
// MODE 2
// MAZE
// =====================================================

void maze()
{
  showMode();

  motorControl(0, 0);

  delay(500);

  while (true)
  {
    receiveData();

    // CENTER CLEAR
    if (centerDistance > obstacleDistance)
    {
      motorControl(
        forwardSpeed,
        forwardSpeed
      );

      continue;
    }


    // CENTER BLOCKED
    motorControl(0, 0);

    delay(100);


    // RIGHT OPEN
    if (rightDistance > obstacleDistance)
    {
      turnRight();

      delay(350);

      motorControl(0, 0);

      delay(100);

      continue;
    }


    // LEFT OPEN
    if (leftDistance > obstacleDistance)
    {
      turnLeft();

      delay(350);

      motorControl(0, 0);

      delay(100);

      continue;
    }


    // DEAD END
    uTurn();

    delay(700);

    motorControl(0, 0);

    delay(100);
  }
}


// =====================================================
// MODE 3
// RC MANUAL
// =====================================================

void rcManual()
{
  showMode();

  motorControl(0, 0);

  delay(300);

  while (true)
  {
    receiveData();

    int throttle =
      map(
        rcThrottle,
        1000,
        2000,
        -255,
        255
      );

    int steering =
      map(
        rcSteering,
        1000,
        2000,
        -180,
        180
      );


    // Dead zones
    if (abs(throttle) < 25)
      throttle = 0;

    if (abs(steering) < 20)
      steering = 0;


    // Differential drive
    int leftMotor =
      throttle + steering;

    int rightMotor =
      throttle - steering;


    leftMotor =
      constrain(
        leftMotor,
        -255,
        255
      );

    rightMotor =
      constrain(
        rightMotor,
        -255,
        255
      );


    motorControl(
      leftMotor,
      rightMotor
    );
  }
}


// =====================================================
// IR SENSOR READING
// =====================================================

void readSensors()
{
  lineFound = false;

  for (int i = 0; i < 8; i++)
  {
    int rawValue =
      analogRead(i);

    analog[i] =
      map(
        rawValue,
        minimum[i],
        maximum[i],
        0,
        1000
      );

    analog[i] =
      constrain(
        analog[i],
        0,
        1000
      );

    binary[i] =
      analog[i] > 500;

    if (binary[i])
      lineFound = true;
  }
}


// =====================================================
// PID
// =====================================================

void movePID()
{
  error = 0;
  sensorCount = 0;

  for (int i = 0; i < 8; i++)
  {
    error +=
      weights[i] * binary[i];

    sensorCount +=
      binary[i];
  }

  if (sensorCount > 0)
    error /= sensorCount;


  P = error;

  I += error;

  I = constrain(I, -100, 100);

  D = error - lastError;

  PIDoutput =
    kp * P +
    ki * I +
    kd * D;

  lastError = error;


  speedA =
    robotSpeed - PIDoutput;

  speedB =
    robotSpeed + PIDoutput;


  speedA =
    constrain(speedA, -255, 255);

  speedB =
    constrain(speedB, -255, 255);


  motorControl(
    speedA,
    speedB
  );
}


// =====================================================
// CALIBRATION
// =====================================================

void calibrate()
{
  for (int j = 0; j < 2000; j++)
  {
    motorControl(
      calibrationSpeed,
      -calibrationSpeed
    );

    for (int i = 0; i < 8; i++)
    {
      int val =
        analogRead(i);

      if (val < minimum[i])
        minimum[i] = val;

      if (val > maximum[i])
        maximum[i] = val;
    }

    receiveData();
  }

  motorControl(0, 0);
}


// =====================================================
// MOTOR CONTROL
// =====================================================

void motorControl(
  int speedA,
  int speedB
)
{
  // LEFT
  if (speedA > 0)
  {
    digitalWrite(AIN1, HIGH);
    digitalWrite(AIN2, LOW);

    analogWrite(PWMA, speedA);
  }
  else if (speedA < 0)
  {
    digitalWrite(AIN1, LOW);
    digitalWrite(AIN2, HIGH);

    analogWrite(PWMA, abs(speedA));
  }
  else
  {
    digitalWrite(AIN1, HIGH);
    digitalWrite(AIN2, HIGH);

    analogWrite(PWMA, 0);
  }


  // RIGHT
  if (speedB > 0)
  {
    digitalWrite(BIN1, LOW);
    digitalWrite(BIN2, HIGH);

    analogWrite(PWMB, speedB);
  }
  else if (speedB < 0)
  {
    digitalWrite(BIN1, HIGH);
    digitalWrite(BIN2, LOW);

    analogWrite(PWMB, abs(speedB));
  }
  else
  {
    digitalWrite(BIN1, HIGH);
    digitalWrite(BIN2, HIGH);

    analogWrite(PWMB, 0);
  }
}


// =====================================================
// TURNS
// =====================================================

void turnLeft()
{
  motorControl(
    -backwardSpeed,
    forwardSpeed
  );
}

void turnRight()
{
  motorControl(
    forwardSpeed,
    -backwardSpeed
  );
}

void uTurn()
{
  motorControl(
    -uturnSpeed,
    uturnSpeed
  );
}

// =====================================================
// NANO 2
// ULTRASONIC + RC RECEIVER
//
// HC-SR04:
// LEFT    D2 TRIG / D3 ECHO
// CENTER  D4 TRIG / D5 ECHO
// RIGHT   D6 TRIG / D7 ECHO
//
// RC RECEIVER:
// CH1 STEERING -> D8
// CH2 THROTTLE -> D9
//
// TX D1 -> NANO 1 RX D0
// =====================================================

#define TRIG_LEFT    2
#define ECHO_LEFT    3

#define TRIG_CENTER  4
#define ECHO_CENTER  5

#define TRIG_RIGHT   6
#define ECHO_RIGHT   7

#define RC_STEERING  8
#define RC_THROTTLE  9


// =====================================================
// ULTRASONIC
// =====================================================

long getDistance(
  int trigPin,
  int echoPin
)
{
  digitalWrite(
    trigPin,
    LOW
  );

  delayMicroseconds(2);

  digitalWrite(
    trigPin,
    HIGH
  );

  delayMicroseconds(10);

  digitalWrite(
    trigPin,
    LOW
  );

  long duration =
    pulseIn(
      echoPin,
      HIGH,
      25000
    );

  if (duration == 0)
    return 400;

  long distance =
    duration * 0.0343 / 2;

  if (distance > 400)
    distance = 400;

  return distance;
}


// =====================================================
// SETUP
// =====================================================

void setup()
{
  Serial.begin(9600);

  pinMode(TRIG_LEFT, OUTPUT);
  pinMode(ECHO_LEFT, INPUT);

  pinMode(TRIG_CENTER, OUTPUT);
  pinMode(ECHO_CENTER, INPUT);

  pinMode(TRIG_RIGHT, OUTPUT);
  pinMode(ECHO_RIGHT, INPUT);

  pinMode(
    RC_STEERING,
    INPUT
  );

  pinMode(
    RC_THROTTLE,
    INPUT
  );

  digitalWrite(TRIG_LEFT, LOW);
  digitalWrite(TRIG_CENTER, LOW);
  digitalWrite(TRIG_RIGHT, LOW);
}


// =====================================================
// LOOP
// =====================================================

void loop()
{
  // Ultrasonic
  long left =
    getDistance(
      TRIG_LEFT,
      ECHO_LEFT
    );

  delay(20);

  long center =
    getDistance(
      TRIG_CENTER,
      ECHO_CENTER
    );

  delay(20);

  long right =
    getDistance(
      TRIG_RIGHT,
      ECHO_RIGHT
    );


  // RC receiver
  int steering =
    pulseIn(
      RC_STEERING,
      HIGH,
      25000
    );

  int throttle =
    pulseIn(
      RC_THROTTLE,
      HIGH,
      25000
    );


  // If signal is lost, use neutral
  if (steering < 900 || steering > 2100)
    steering = 1500;

  if (throttle < 900 || throttle > 2100)
    throttle = 1500;


  // Send:
  // LEFT,CENTER,RIGHT,STEERING,THROTTLE

  Serial.print(left);
  Serial.print(",");

  Serial.print(center);
  Serial.print(",");

  Serial.print(right);
  Serial.print(",");

  Serial.print(steering);
  Serial.print(",");

  Serial.println(throttle);


  delay(20);
}
