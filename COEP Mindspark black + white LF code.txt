// ============================================================
// DUAL-SURFACE LINE FOLLOWER
// HALF 1 : BLACK LINE + WHITE FLOOR
// HALF 2 : WHITE LINE + BLACK FLOOR
// ============================================================
//
// LEFT BUTTON  -> CALIBRATION
// RIGHT BUTTON -> START RUN
//
// During calibration the robot rotates clockwise.
//
// IMPORTANT:
// The robot starts in BLACK-LINE mode.
// When it detects a wide BLACK area (all sensors dark),
// it switches permanently to WHITE-LINE mode.
//
// ============================================================


// ---------------- MOTOR PINS ----------------

#define PWMB 9
#define PWMA 3

#define BIN1 7
#define BIN2 8

#define AIN1 5
#define AIN2 4

#define STDBY 6


// ---------------- PERIPHERALS ----------------

#define led1 11
#define led2 12

#define Button1 2
#define Button2 10


// ============================================================
// PID VARIABLES
// ============================================================

int P, D, I, lastError, PIDoutput;
int error;

int speedA, speedB;
int sensorCount;

int robotSpeed = 230;
int forwardSpeed = 200;
int backwardSpeed = 230;

int calibrationSpeed = 100;
int uturnSpeed = 120;

int kp = 30;
int kd = 120;
int ki = 0;


// ============================================================
// SENSOR VARIABLES
// ============================================================

int weights[8] = {
  -8, -4, -2, -1,
   1,  2,  4,  8
};

int minimum[8] = {
  1023, 1023, 1023, 1023,
  1023, 1023, 1023, 1023
};

int maximum[8] = {
  0, 0, 0, 0,
  0, 0, 0, 0
};

int analog[8];
int binary[8];

bool lineFound = false;


// ============================================================
// SURFACE MODE
// ============================================================

// 0 = BLACK LINE ON WHITE FLOOR
// 1 = WHITE LINE ON BLACK FLOOR

int lineMode = 0;


// ============================================================
// TRANSITION DETECTION
// ============================================================

// Number of consecutive readings required
// before switching to WHITE mode.

int transitionCount = 0;

#define TRANSITION_LIMIT 15


// ============================================================
// SETUP
// ============================================================

void setup() {

  Serial.begin(9600);

  pinMode(STDBY, OUTPUT);

  pinMode(AIN1, OUTPUT);
  pinMode(AIN2, OUTPUT);

  pinMode(BIN1, OUTPUT);
  pinMode(BIN2, OUTPUT);

  pinMode(PWMA, OUTPUT);
  pinMode(PWMB, OUTPUT);

  pinMode(led1, OUTPUT);
  pinMode(led2, OUTPUT);

  pinMode(Button1, INPUT_PULLUP);
  pinMode(Button2, INPUT_PULLUP);

  digitalWrite(STDBY, HIGH);


  // ----------------------------------------------------------
  // WAIT FOR CALIBRATION BUTTON
  // ----------------------------------------------------------

  while (digitalRead(Button1));

  delay(100);

  digitalWrite(led1, HIGH);

  delay(1000);

  calibrate();

  delay(100);

  digitalWrite(led1, LOW);


  // ----------------------------------------------------------
  // WAIT FOR RUN BUTTON
  // ----------------------------------------------------------

  while (digitalRead(Button2));

  delay(1000);

  digitalWrite(led2, HIGH);
}


// ============================================================
// MAIN LOOP
// ============================================================

void loop() {

  readSensors();

  // ----------------------------------------------------------
  // CHECK WHETHER ROBOT HAS ENTERED SECOND HALF
  // ----------------------------------------------------------

  checkTransition();


  // ----------------------------------------------------------
  // NORMAL LINE FOLLOWING
  // ----------------------------------------------------------

  if (lineFound) {

    move();

  }

  else {

    // No line detected

    if (lastError > 0) {

      turnLeft();

    }

    else if (lastError < 0) {

      turnRight();

    }

    else {

      uTurn();

    }
  }
}


// ============================================================
// READ SENSORS
// ============================================================

void readSensors() {

  lineFound = false;

  for (int i = 0; i < 8; i++) {

    analog[i] = map(
      analogRead(i),
      minimum[i],
      maximum[i],
      0,
      1000
    );

    analog[i] = constrain(
      analog[i],
      0,
      1000
    );


    // ========================================================
    // BLACK LINE MODE
    //
    // BLACK LINE = LOW SENSOR VALUE
    // ========================================================

    if (lineMode == 0) {

      if (analog[i] < 500) {

        binary[i] = 1;
        lineFound = true;

      }

      else {

        binary[i] = 0;

      }
    }


    // ========================================================
    // WHITE LINE MODE
    //
    // WHITE LINE = HIGH SENSOR VALUE
    // ========================================================

    else {

      if (analog[i] > 500) {

        binary[i] = 1;
        lineFound = true;

      }

      else {

        binary[i] = 0;

      }
    }
  }
}


// ============================================================
// DETECT TRANSITION FROM BLACK SECTION TO WHITE SECTION
// ============================================================

void checkTransition() {

  // Only look for transition while in BLACK mode

  if (lineMode != 0) {
    return;
  }


  int darkSensors = 0;


  // Count sensors seeing dark surface

  for (int i = 0; i < 8; i++) {

    if (analog[i] < 300) {

      darkSensors++;

    }
  }


  // ----------------------------------------------------------
  // If almost/all sensors see black:
  // robot may have entered black floor of second section
  // ----------------------------------------------------------

  if (darkSensors >= 7) {

    transitionCount++;

  }

  else {

    transitionCount = 0;

  }


  // ----------------------------------------------------------
  // Confirm transition
  // ----------------------------------------------------------

  if (transitionCount >= TRANSITION_LIMIT) {

    lineMode = 1;

    transitionCount = 0;

    // Reset PID
    I = 0;
    lastError = 0;

    Serial.println("WHITE LINE MODE");

    // Small delay to prevent repeated switching
    delay(100);

  }
}


// ============================================================
// PID LINE FOLLOWING
// ============================================================

void move() {

  error = 0;

  sensorCount = 0;


  for (int i = 0; i < 8; i++) {

    if (binary[i]) {

      error += weights[i];

      sensorCount++;

    }
  }


  // Safety
  if (sensorCount == 0) {

    return;

  }


  error = error / sensorCount;


  // ---------------- PID ----------------

  P = error;

  I = I + error;

  I = constrain(I, -100, 100);

  D = error - lastError;


  PIDoutput =
    (kp * P) +
    (ki * I) +
    (kd * D);


  lastError = error;


  // ---------------- MOTOR SPEED ----------------

  speedA = robotSpeed - PIDoutput;
  speedB = robotSpeed + PIDoutput;


  speedA = constrain(
    speedA,
    -255,
    255
  );

  speedB = constrain(
    speedB,
    -255,
    255
  );


  motorControl(
    speedA,
    speedB
  );
}


// ============================================================
// CALIBRATION
// ============================================================

void calibrate() {

  for (int j = 0; j < 2000; j++) {

    // Rotate clockwise

    motorControl(
      calibrationSpeed,
      -calibrationSpeed
    );


    for (int i = 0; i < 8; i++) {

      int val = analogRead(i);


      if (val < minimum[i]) {

        minimum[i] = val;

      }


      if (val > maximum[i]) {

        maximum[i] = val;

      }
    }
  }


  motorControl(0, 0);
}


// ============================================================
// MOTOR CONTROL
// ============================================================

void motorControl(
  int speedA,
  int speedB
) {


  // ==========================================================
  // MOTOR A
  // ==========================================================

  if (speedA > 0) {

    digitalWrite(AIN1, HIGH);
    digitalWrite(AIN2, LOW);

    analogWrite(
      PWMA,
      speedA
    );

  }

  else if (speedA < 0) {

    digitalWrite(AIN1, LOW);
    digitalWrite(AIN2, HIGH);

    analogWrite(
      PWMA,
      abs(speedA)
    );

  }

  else {

    digitalWrite(AIN1, HIGH);
    digitalWrite(AIN2, HIGH);

    analogWrite(PWMA, 0);
  }


  // ==========================================================
  // MOTOR B
  // ==========================================================

  if (speedB > 0) {

    digitalWrite(BIN1, HIGH);
    digitalWrite(BIN2, LOW);

    analogWrite(
      PWMB,
      speedB
    );

  }

  else if (speedB < 0) {

    digitalWrite(BIN1, LOW);
    digitalWrite(BIN2, HIGH);

    analogWrite(
      PWMB,
      abs(speedB)
    );

  }

  else {

    digitalWrite(BIN1, HIGH);
    digitalWrite(BIN2, HIGH);

    analogWrite(PWMB, 0);
  }
}


// ============================================================
// LEFT TURN
// ============================================================

void turnLeft() {

  motorControl(
    -backwardSpeed,
    forwardSpeed
  );
}


// ============================================================
// RIGHT TURN
// ============================================================

void turnRight() {

  motorControl(
    forwardSpeed,
    -backwardSpeed
  );
}


// ============================================================
// U-TURN
// ============================================================

void uTurn() {

  motorControl(
    -uturnSpeed,
    uturnSpeed
  );
}
