// USE LEFT BUTTON TO START CALIBERATION, ON BUTTON PRESS, LED1 WILL GLOW, BOT SHOULD ROTATE CLOCKWISE AND LED1 WILL TURN OFF.
// USE RIGHT BUTTON TO START THE RUN, LED 2 WILL TURN ON AND WILL REMAIN ON.

// Motor Pins
#define PWMB 9
#define PWMA 3
#define BIN1 7
#define BIN2 8
#define AIN1 5
#define AIN2 4
#define STDBY 6

// Peripherals
#define led1 11
#define led2 12
#define Button1 2
#define Button2 10

// Code Variables
int P, D, I, lastError, PIDoutput;
int error;
int speedA, speedB;
int sensorCount;
int robotSpeed = 230;
int forwardSpeed = 200;
int backwardSpeed = 230;
int calibrationSpeed = 100;
int uturnSpeed = 120;
int kp = 30;
int kd = 120;
int ki = 0;
bool lineFound = true;
int weights[8] = {-8,-4,-2,-1,1,2,4,8};
int minimum[8]= {1023,1023,1023,1023,1023,1023,1023,1023};
int maximum[8] = {0,0,0,0,0,0,0,0};
int analog[8], binary[8];

void setup() {
  Serial.begin(9600);
  pinMode(STDBY, OUTPUT);
  pinMode(AIN1, OUTPUT);
  pinMode(AIN2, OUTPUT);
  pinMode(BIN1, OUTPUT);
  pinMode(BIN2, OUTPUT);
  pinMode(PWMA, OUTPUT);
  pinMode(PWMB, OUTPUT);
  pinMode(led2, OUTPUT);
  pinMode(led1, OUTPUT);
  pinMode(Button1, INPUT_PULLUP);
  pinMode(Button2, INPUT_PULLUP);
  digitalWrite(STDBY,HIGH);
  while(digitalRead(Button1));
  delay(100);
  digitalWrite(led1,1);
  delay(1000);
  calibrate();
  delay(100);
  digitalWrite(led1,0);
  while(digitalRead(Button2));
  delay(1000);
  digitalWrite(led2,1);
}

void loop() {
  readSensors();
  if (lineFound) {
    move();
  }
  else {
    if (error > 0 ) {
      turnLeft();
    }
    else if (error < 0) {
      turnRight();
    }
    else {
      uTurn();
    }
  }
}

void readSensors() {
  lineFound = false;
  for (int i = 0; i < 8; i++) {
    analog[i] = map(analogRead(i), minimum[i], maximum[i], 0, 1000);
    analog[i] = constrain(analog[i], 0, 1000);
    binary[i] = analog[i] > 500;
    if (binary[i]) lineFound = true;
  }
}

void move() {
  error = 0;
  sensorCount = 0;
  for (int i = 0; i < 8; i++) {
    error += weights[i] * binary[i];
    sensorCount += binary[i];
  }
  error = error / sensorCount;
  P = error;
  I = I + error;
  D = error - lastError;
  PIDoutput = (kp * P) + (ki * I) + (kd * D);
  lastError = error;
  speedA = robotSpeed - PIDoutput;
  speedB = robotSpeed + PIDoutput;
  speedA = constrain(speedA, -255, 255);
  speedB = constrain(speedB, -255, 255);
  motorControl(speedA,speedB);
}

void calibrate() {
  for (int j = 0; j < 2000; j++) {
   motorControl(calibrationSpeed,-calibrationSpeed);
    for (int i = 0; i < 8; i++) {
      int val = analogRead(i);
      if (val < minimum[i]) {
        minimum[i] = val;
      }
      if (val > maximum[i]) {
        maximum[i] = val;
      }
    }
  }
  motorControl(0,0);
}

void motorControl(int speedA,int speedB){
 if (speedA > 0) {
  digitalWrite(AIN1, 1);
  digitalWrite(AIN2, 0);
  analogWrite(PWMA, speedA);
 }
 else if (speedA < 0) {
  digitalWrite(AIN1, 0);
  digitalWrite(AIN2, 1);
  analogWrite(PWMA, abs(speedA));
 }
 else {
  digitalWrite(AIN1, 1);
  digitalWrite(AIN2, 1);
  analogWrite(PWMA, 0);
 }
 if (speedB > 0) {
  digitalWrite(BIN1, 1);
  digitalWrite(BIN2, 0);
  analogWrite(PWMB, speedB);
 }
 else if (speedB < 0) {
  digitalWrite(BIN1, 0);
  digitalWrite(BIN2, 1);
  analogWrite(PWMB, abs(speedB));
 }
 else {
  digitalWrite(BIN1, 1);
  digitalWrite(BIN2, 1);
  analogWrite(PWMB, 0);
 }
}
void turnLeft(){
  motorControl(-backwardSpeed,forwardSpeed);
}
void turnRight(){
  motorControl(forwardSpeed,-backwardSpeed);
}
void uTurn(){
  motorControl(-uturnSpeed,uturnSpeed);
}
