char pass[] = "88888888";

// L298N motor driver pins
#define ENA 25
#define IN1 26
#define IN2 27

#define ENB 33
#define IN3 18
#define IN4 19

// Maximum PWM command
const int motorSpeed = 255;

// Slightly increased turning settings
const int steeringLimit = 170;
const int turnSpeed = 240;

bool carStarted = false;

// Button states
bool forwardPressed = false;
bool backwardPressed = false;
bool leftPressed = false;
bool rightPressed = false;

// Stop motors
void stopMotors() {
  digitalWrite(IN1, LOW);
  digitalWrite(IN2, LOW);
  digitalWrite(IN3, LOW);
  digitalWrite(IN4, LOW);

  analogWrite(ENA, 0);
  analogWrite(ENB, 0);
}

// Reset direction buttons
void resetButtons() {
  forwardPressed = false;
  backwardPressed = false;
  leftPressed = false;
  rightPressed = false;
}

// Set motor direction and speed
void setMotor(int leftSpeed, int rightSpeed) {
  leftSpeed = constrain(leftSpeed, -255, 255);
  rightSpeed = constrain(rightSpeed, -255, 255);

  // Left motor
  if (leftSpeed > 0) {
    digitalWrite(IN1, HIGH);
    digitalWrite(IN2, LOW);
  } else if (leftSpeed < 0) {
    digitalWrite(IN1, LOW);
    digitalWrite(IN2, HIGH);
  } else {
    digitalWrite(IN1, LOW);
    digitalWrite(IN2, LOW);
  }

  // Right motor
  if (rightSpeed > 0) {
    digitalWrite(IN3, HIGH);
    digitalWrite(IN4, LOW);
  } else if (rightSpeed < 0) {
    digitalWrite(IN3, LOW);
    digitalWrite(IN4, HIGH);
  } else {
    digitalWrite(IN3, LOW);
    digitalWrite(IN4, LOW);
  }

  analogWrite(ENA, abs(leftSpeed));
  analogWrite(ENB, abs(rightSpeed));
}

// Update movement from all button states
void updateMotors() {
  if (!carStarted) {
    stopMotors();
    return;
  }

  int throttle =
      (int)forwardPressed - (int)backwardPressed;

  int steering =
      (int)rightPressed - (int)leftPressed;

  // No movement command
  if (throttle == 0 && steering == 0) {
    stopMotors();
    return;
  }

  int leftSpeed;
  int rightSpeed;

  // Turn in place when only left or right is pressed
  if (throttle == 0 && steering != 0) {
    leftSpeed = steering * turnSpeed;
    rightSpeed = -steering * turnSpeed;
  } else {
    // Full forward/reverse speed with steering
    leftSpeed = throttle * motorSpeed
                + steering * steeringLimit;

    rightSpeed = throttle * motorSpeed
                 - steering * steeringLimit;

    // Preserve the speed ratio without exceeding PWM 255
    int maxSpeed = max(abs(leftSpeed), abs(rightSpeed));

    if (maxSpeed > 255) {
      leftSpeed = (long)leftSpeed * 255 / maxSpeed;
      rightSpeed = (long)rightSpeed * 255 / maxSpeed;
    }
  }

  setMotor(leftSpeed, rightSpeed);
}

// V0: Forward
BLYNK_WRITE(V0) {
  forwardPressed = param.asInt();
  updateMotors();
}

// V1: Backward
BLYNK_WRITE(V1) {
  backwardPressed = param.asInt();
  updateMotors();
}

// V2: Left
BLYNK_WRITE(V2) {
  leftPressed = param.asInt();
  updateMotors();
}

// V3: Right
BLYNK_WRITE(V3) {
  rightPressed = param.asInt();
  updateMotors();
}

// V4: Stop
BLYNK_WRITE(V4) {
  if (param.asInt()) {
    resetButtons();
    stopMotors();

    Blynk.virtualWrite(V0, 0);
    Blynk.virtualWrite(V1, 0);
    Blynk.virtualWrite(V2, 0);
    Blynk.virtualWrite(V3, 0);
  }
}

// V5: Start / Enable
BLYNK_WRITE(V5) {
  carStarted = (param.asInt() == 1);

  if (!carStarted) {
    resetButtons();
    stopMotors();

    Blynk.virtualWrite(V0, 0);
    Blynk.virtualWrite(V1, 0);
    Blynk.virtualWrite(V2, 0);
    Blynk.virtualWrite(V3, 0);
  } else {
    updateMotors();
  }
}

// Blynk connection handler
BLYNK_CONNECTED() {
  carStarted = false;
  resetButtons();
  stopMotors();

  Blynk.virtualWrite(V0, 0);
  Blynk.virtualWrite(V1, 0);
  Blynk.virtualWrite(V2, 0);
  Blynk.virtualWrite(V3, 0);
  Blynk.virtualWrite(V4, 0);
  Blynk.virtualWrite(V5, 0);
}

// Setup
void setup() {
  Serial.begin(115200);

  pinMode(ENA, OUTPUT);
  pinMode(IN1, OUTPUT);
  pinMode(IN2, OUTPUT);

  pinMode(ENB, OUTPUT);
  pinMode(IN3, OUTPUT);
  pinMode(IN4, OUTPUT);

  stopMotors();

  Serial.println("Connecting to Blynk...");
  Blynk.begin(BLYNK_AUTH_TOKEN, ssid, pass);

  Serial.println("RC Car Ready");
}

// Main loop
void loop() {
  Blynk.run();
}
